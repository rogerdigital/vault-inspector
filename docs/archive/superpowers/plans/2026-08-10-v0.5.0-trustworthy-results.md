# Vault Inspector 0.5.0 Trustworthy Results Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Turn the existing scan result list into a trustworthy maintenance loop that explains finding certainty, compares same-profile scans, and verifies user actions.

**Architecture:** Keep the eight scanners and `ScanRunner` as the detection boundary. Add required presentation metadata to `Issue`, pure scan-profile and fingerprint-diff modules, one validated successful-scan snapshot in plugin data, and report-only lifecycle state layered over the raw `ScanResult`. The Obsidian plugin owns snapshot persistence and action verification; the CLI receives additive finding metadata but keeps its existing baseline and exit-code behavior.

**Tech Stack:** TypeScript, Obsidian Plugin API, esbuild, Vitest, ESLint, CSS

**Design:** `docs/superpowers/specs/2026-08-09-v0.5.0-trustworthy-results-design.md`

---

## File map

### Finding semantics

- Modify `src/scanner/Issue.ts`: define classification, explanation, lifecycle,
  and snapshot-facing issue contracts.
- Create `src/scanner/finding-presentation.ts`: construct consistent finding
  presentation metadata.
- Modify each file in `src/scanner/scanners/`: attach scanner-specific metadata
  without changing fingerprints or detection.
- Modify the eight scanner test files in `src/tests/`: lock classification and
  explanation behavior.

### Scan comparison and persistence

- Create `src/scanner/scan-profile.ts`: canonicalize detection settings into a
  deterministic profile signature.
- Create `src/scanner/result-diff.ts`: calculate lifecycle status by fingerprint
  in linear time.
- Create `src/snapshot/scan-snapshot.ts`: snapshot types, validation, and
  conversion from `ScanResult`.
- Create `src/settings/plugin-data.ts`: parse legacy flat settings and the new
  settings-plus-snapshot envelope.
- Modify `src/main.ts`: load/save the envelope, accept successful results, and
  replace snapshots only after successful scans.

### Report experience

- Modify `src/report/report-model.ts`: add lifecycle/classification filters,
  deterministic ordering, comparison state, and action feedback state.
- Modify `src/report/render-summary.ts`: show Active/New/Persisting/Resolved
  counts or a comparison-unavailable reason.
- Create `src/report/render-evidence.ts`: render classification and the
  What/Why/Caveat/Next contract.
- Create `src/report/render-changes.ts`: render non-actionable resolved findings.
- Modify `src/report/render-issues.ts`: add lifecycle badges, evidence disclosure,
  and contextual actions.
- Modify `src/report/InspectorView.ts`: wire filters, resolved state, contextual
  actions, and outcome panels.
- Modify `styles.css`: style new badges, disclosures, panels, and narrow layouts.

### Action feedback and compatibility

- Create `src/fix/action-outcomes.ts`: represent and summarize mutation and
  disposition outcomes.
- Create `src/report/render-outcomes.ts`: display per-item batch results.
- Create `src/report/exclude-folder-modal.ts`: confirm scanner-specific parent
  folder exclusions.
- Modify `src/main.ts`: preserve per-item errors and verify mutation results.
- Modify `src/report/markdown-export.ts`: include classification and explanation.
- Modify `src/tests/cli.test.ts`: prove additive CLI JSON fields do not change
  schema, baseline, or exit codes.
- Modify `README.md` and `skills/vault-inspector/SKILL.md`: document the shipped

```ts
import type {
	FindingClassification,
	FindingExplanation,
} from "./Issue";

export type FindingPresentation = {
	classification: FindingClassification;
	explanation: FindingExplanation;
};

export function describeFinding(
	classification: FindingClassification,
	why: string,
	nextStep: string,
	caveat?: string,
): FindingPresentation {
	return {
		classification,
		explanation: {
			why,
			...(caveat === undefined ? {} : { caveat }),
			nextStep,
		},
	};
}
```

```ts
import { SCANNER_IDS } from "./Issue";
import type { InspectorSettings } from "../settings/settings";

function normalized(values: string[]): string[] {
	return [...new Set(values)].sort();
}

function hashString(value: string): string {
	let first = 0x811c9dc5;
	let second = 0x01000193;
	for (let index = 0; index < value.length; index++) {
		const code = value.charCodeAt(index);
		first = ((first << 5) - first + code) | 0;
		second = ((second << 5) - second + code) | 0;
	}
	return (first >>> 0).toString(36) + (second >>> 0).toString(36);
}

export function createScanProfile(settings: InspectorSettings): string {
	const canonical = {
		enabledScanners: SCANNER_IDS.map((id) => [id, settings.enabledScanners[id]]),
		ignoredFolders: normalized(settings.ignoredFolders),
		ignoredFoldersByScanner: SCANNER_IDS.map((id) => [
			id,
			normalized(settings.ignoredFoldersByScanner[id] ?? []),
		]),
		largeMarkdownBytes: settings.largeMarkdownBytes,
		largeAttachmentBytes: settings.largeAttachmentBytes,
		ignoredLargeMarkdownFrontmatterKeys: normalized(
			settings.ignoredLargeMarkdownFrontmatterKeys,
		),
		ignoredLargeMarkdownPathPatterns: normalized(
			settings.ignoredLargeMarkdownPathPatterns,
		),
		duplicateHashMaxBytes: settings.duplicateHashMaxBytes,
		lowUsageTagThreshold: settings.lowUsageTagThreshold,
		emptyNoteWordThreshold: settings.emptyNoteWordThreshold,
		watchedTags: normalized(settings.watchedTags),
		ignoredProperties: normalized(settings.ignoredProperties),
	};

	return hashString(JSON.stringify(canonical));
}
```

```ts
import type { ScanResult } from "./Issue";
import {
	COMPARISON_VERSION,
	type ScanSnapshot,
	type SnapshotIssue,
} from "../snapshot/scan-snapshot";

export type CurrentFindingStatus = "new" | "persisting";
export type ComparisonUnavailableReason =
	| "first-scan"
	| "settings-changed"
	| "semantics-changed";

export type LifecycleComparison = {
	available: boolean;
	reason?: ComparisonUnavailableReason;
	statuses: Map<string, CurrentFindingStatus>;
	resolvedIssues: SnapshotIssue[];
};

export function compareScanResult(
	current: ScanResult,
	snapshot: ScanSnapshot | null,
	currentProfile: string,
): LifecycleComparison {
	if (!snapshot) {
		return { available: false, reason: "first-scan", statuses: new Map(), resolvedIssues: [] };
	}
	if (snapshot.comparisonVersion !== COMPARISON_VERSION) {
		return { available: false, reason: "semantics-changed", statuses: new Map(), resolvedIssues: [] };
	}
	if (snapshot.scanProfile !== currentProfile) {
		return { available: false, reason: "settings-changed", statuses: new Map(), resolvedIssues: [] };
	}

	const previous = new Map(snapshot.issues.map((issue) => [issue.fingerprint, issue]));
	const currentIssues = [...current.issues, ...current.ignoredIssues];
	const currentFingerprints = new Set(currentIssues.map((issue) => issue.fingerprint));
	const statuses = new Map<string, CurrentFindingStatus>();
	for (const issue of currentIssues) {
		statuses.set(issue.fingerprint, previous.has(issue.fingerprint) ? "persisting" : "new");
	}
	const resolvedIssues = snapshot.issues.filter(
		(issue) => !currentFingerprints.has(issue.fingerprint),
	);
	return { available: true, statuses, resolvedIssues };
}
```

execution error handling, and one final verification scan:

```ts
import type { FixAction, Issue, ScanResult } from "../scanner/Issue";
import {
	getFreshFixAction,
	type FixDecision,
} from "./fix-decisions";
import type { ActionOutcome } from "./action-outcomes";

type PendingAction = {
	index: number;
	fingerprint: string;
	affectedPaths: string[];
	affectedCount: number;
};

export async function runFixBatch(
	issues: Issue[],
	decisions: FixDecision[],
	dependencies: FixRunnerDependencies,
): Promise<FixBatchResult> {
	const decisionsByFingerprint = new Map(
		decisions.map((decision) => [decision.fingerprint, decision]),
	);
	const outcomes: Array<ActionOutcome | null> = issues.map(() => null);
	const pending: PendingAction[] = [];

	for (const [index, issue] of issues.entries()) {
		const decision = decisionsByFingerprint.get(issue.fingerprint);
		if (!decision) {
			outcomes[index] = skipped(issue, "No confirmed fix decision was available.");
			continue;
		}

		const freshResult = await dependencies.scan();
		const freshIssue = freshResult
			? [...freshResult.issues, ...freshResult.ignoredIssues].find(
				(candidate) => candidate.fingerprint === issue.fingerprint,
			)
			: undefined;
		const action = getFreshFixAction(issue, freshIssue, decision);
		if (!action) {
			outcomes[index] = skipped(
				issue,
				freshResult
					? "The finding or fix evidence changed before execution."
					: "The preflight scan did not complete.",
			);
			continue;
		}

		try {
			pending.push({
				index,
				fingerprint: issue.fingerprint,
				affectedPaths: [...action.targetPaths],
				affectedCount: await dependencies.execute(action),
			});
		} catch (error) {
			outcomes[index] = {
				fingerprint: issue.fingerprint,
				outcome: "failed",
				phase: "execution",
				message: error instanceof Error ? error.message : String(error),
				affectedPaths: [...action.targetPaths],
			};
		}
	}

	const verificationResult = await dependencies.scan();
	if (!verificationResult) {
		for (const action of pending) {
			outcomes[action.index] = {
				fingerprint: action.fingerprint,
				outcome: "failed",
				phase: "verification",
				message: "The final verification scan did not complete.",
				affectedPaths: action.affectedPaths,
			};
		}
	} else {
		const remaining = new Set([
			...verificationResult.issues,
			...verificationResult.ignoredIssues,
		].map((issue) => issue.fingerprint));
		for (const action of pending) {
			const stillPresent = remaining.has(action.fingerprint);
			outcomes[action.index] = {
				fingerprint: action.fingerprint,
				outcome: stillPresent ? "still-present" : "fixed",
				message: stillPresent
					? `The finding remains after ${action.affectedCount} change(s).`
					: `Verified after ${action.affectedCount} change(s).`,
				affectedPaths: action.affectedPaths,
			};
		}
	}

	return {
		outcomes: outcomes.filter((outcome): outcome is ActionOutcome => outcome !== null),
		verificationResult,
	};
}

function skipped(issue: Issue, message: string): ActionOutcome {
	return {
		fingerprint: issue.fingerprint,
		outcome: "skipped",
		phase: "preflight",
		message,
		affectedPaths: issue.fixAction?.targetPaths ?? [],
	};
}
```

Never catch and discard an exception. `scan()` already turns a displayed scan
failure into `null`; if that boundary later changes to reject, update the
dependency wrapper in `main.ts` to convert the rejection into the same visible
preflight or verification failure.
